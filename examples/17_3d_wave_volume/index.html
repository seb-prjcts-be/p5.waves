Example 17 — WEBGL · 3D

3D Wave Volume

16×16 grid with three independent createSampler() calls — one per axis, each with shift: true on its own cycle. Y drives the surface height, X and Z warp the grid sideways and in depth. Points near the surface glow; the rest stay hidden. Drag to rotate.

Core snippet
samplerY = Waves.createSampler({
  shift: true,
  shiftInterval: 3,
  range: [-6, 6],
  frequency: 0.4,
  seed: 0
});
samplerX = Waves.createSampler({
  shift: true,
  shiftInterval: 4,
  range: [-3, 3],
  frequency: 0.3,
  seed: 42
});

var dy = samplerY.sample(xi * 0.9 + zi * 0.6, t);
var dx = samplerX.sample(zi * 0.8 + xi * 0.3, t * 0.85);

var surfaceRow = N / 2 + dy;
for (var yi = 0; yi < N; yi++) {
  var gap = Math.abs(yi - surfaceRow);
  if (gap < 2.5) {
    var glow = 1 - gap / 2.5;
    stroke(bright, bright, 255, 255 * glow);
    vertex(x + dx * SPACING * 0.4, y, z);
  }
}
