Example 05 — WEBGL · 3D

Terrain Mesh

Waves.createGrid() produces a Float32Array height map each frame. Rendered as row-by-row TRIANGLE_STRIP in WEBGL, creating an animated 3D terrain that rotates slowly.

Core snippet
// createGrid() → Float32Array height map, updated every sample(t).
// Rendered as TRIANGLE_STRIP rows for a full 3D terrain mesh.

grid = Waves.createGrid(COLS, ROWS, {
  range: [-55, 55],   // Float32Array output
  speed: 0.5
});

// In draw():
const data = grid.sample(frameCount * 0.016);

for (let r = 0; r < ROWS - 1; r++) {
  beginShape(TRIANGLE_STRIP);
  for (let c = 0; c < COLS; c++) {
    vertex(x(c), -data[r * COLS + c],       z(r));
    vertex(x(c), -data[(r+1) * COLS + c],   z(r+1));
  }
  endShape();
}
