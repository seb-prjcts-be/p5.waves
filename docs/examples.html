Waves.wave(y)

One Number

One call returns one number. Use it for size, position, color — anything.

// One number — use it for anything
var v = Waves.wave(i * 0.3, {
  wave: 'bounce', t: t, range: [0, 1]
});

// Row 1: size
circle(x, y, v * 30);

// Row 2: position
circle(x, baseY + v * rowH * 0.8, 8);

// Row 3: color (pure red → pure blue)
fill(255 - round(v * 255), 0, round(v * 255));
circle(x, y, 14);
shift: true

Wave Shift

The wave changes automatically every few seconds. Smooth morph between random formulas. Watch the ribbons transform.

var sampler = Waves.createSampler({
  shift: true,
  amplitude: 1,
  frequency: 0.5
});

// Filled ribbons — color shifts with the wave
var t = millis() / 1000;
for (var i = 0; i < STRIPS; i++) {
  var frac = i / (STRIPS - 1);
  fill(255 - round(frac * 255), 0, round(frac * 255));
  var v = sampler.sample(i * 0.4, t + i * 0.1);
  rect(i * sw, cy - v * rowH, sw - 1, v * rowH * 2);
}

Wild Mode

Left: order. Right: chaos. Same wave, one flag added. The difference speaks for itself.

// Left: stable circles. Right: wild circles.
// Stable — orderly pattern
var sz = Waves.wave(col + row * 0.5, {
  wave: waveIdx, t: t, range: [4, maxR]
});

// Wild — same wave, chaos added
var szW = Waves.wave(col + row * 0.5, {
  wave: waveIdx, t: t, range: [4, maxR],
  mode: 'wild', unpredictability: 0.7

particles + samplers

Flow Fields

Four samplers steer 100 particles. Each sampler uses a different wave. Swap the wave names and the whole flow pattern changes.

// Four samplers, four colors: R, G, B, yellow
var colors = [[255,0,0],[0,255,0],[0,0,255],[255,255,0]];
var samplers = [
  Waves.createSampler({ wave: rw(), range: [-2, 2] }),
  Waves.createSampler({ wave: rw(), range: [-2, 2] }),
  Waves.createSampler({ wave: rw(), range: [-2, 2] }),
  Waves.createSampler({ wave: rw(), range: [-2, 2] })
];

// Each particle follows its assigned sampler
pt.x += samplers[pt.s].sample(pt.y * 0.009, t) * 2.5;
pt.y += samplers[(pt.s+2)%4].sample(pt.x * 0.009, t) * 2.5;
fill(colors[pt.s][0], colors[pt.s][1], colors[pt.s][2], 180);
circle(pt.x, pt.y, 5);

t = any number

Time is a Number

No built-in clock. Pass any number as t. Move the mouse to scrub through time. Freeze it, rewind it, speed it up.

// No clock — t is whatever you pass
// Move mouse to scrub through time
var t = map(mouseX, 0, width, 0, 8);

for (var i = 0; i < 12; i++) {
  var layerT = t + (i / 12) * 6;
  beginShape();
  for (var x = 0; x <= width; x += 4) {
    var dy = Waves.wave(x * 0.015, {
      wave: strataWave, t: layerT,
      amplitude: 15
    });
    vertex(x, y0 + dy);
  }
  endShape();
}

wave → hue + brightness

Color Field

Two waves sample across x and y to control hue and brightness. The result is a flowing color landscape.

// Two shift-samplers drive hue and brightness
var hueS = Waves.createSampler({
  wave: 'meta sine', shift: true,
  shiftInterval: 4, shiftDuration: 2,
  range: [160, 320]
});
var briS = Waves.createSampler({
  wave: 'wobble sine', shift: true,
  shiftInterval: 5, shiftDuration: 2,
  range: [55, 90]
});

// Sample across x and y for a 2D color field
colorMode(HSB, 360, 100, 100);
fill(hueS.sample(x * 0.007 + y * 0.005, t),
     72,
     briS.sample(x * 0.005 - y * 0.007, t));

wave: [a, b], mix

Morph Wave

Blend two wave formulas together. mix: 0 = pure A, mix: 1 = pure B. Animate it and the shape smoothly transforms.

// Blend two formulas with mix
var wA = floor(random(34));
var wB = floor(random(34));

// mix: 0 = pure A, mix: 1 = pure B
var morphMix = (sin(t * 0.3) + 1) * 0.5;

var wv = Waves.wave(x, {
  wave: [wA, wB],    // morph between two waves
  mix: morphMix,
  t: t + row * 0.06,
  frequency: 0.08,
  amplitude: rowH * 0.8
});

Waves.createGrid()

Wave Grid

One call fills a 2D grid. With threshold, each cell is on or off — a binary pattern that animates.

// One call fills a 2D grid
var grid = Waves.createGrid(28, 20, {
  waveRow: 'smooth solid sine',
  waveCol: 'ramp up sine',
  threshold: 0,  // binary: on or off
  speed: 1
});

// Sample once per frame
t += 0.004;
var cells = grid.sample(t);

// Draw: 1 = black, 0 = light
fill(cells[row * 28 + col] === 1 ? 15 : 225);
rect(col * cw, row * ch, cw, ch);
