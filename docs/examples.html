API: Waves.wave(y, options)

The core function

One call returns one number. Pass a position and options — wave shape, time, amplitude, frequency. Use the result as offset, size, colour value, or anything else.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

function setup() {
  createCanvas(600, 400);
  noFill();
}

function draw() {
  background(245);
  const t = frameCount * 0.01;
  const waves = [
    'classic sine', 'triangle',
    'mountain peaks', 'noise'
  ];

  for (let i = 0; i < waves.length; i++) {
    const wave = waves[i];
    const y0 = map(i, 0, 3, 70, height - 70);
    stroke(0, 0, 0, map(i, 0, 3, 220, 80));
    strokeWeight(1.5);
    beginShape();
    for (let x = 0; x <= width; x += 3) {
      vertex(x, y0 + Waves.wave(x * 0.015, { wave, t, amplitude: 45 }));
    }
    endShape();
  }
}

feature: seed · floor(t / interval)

Seed as a time dimension

seed deterministically picks a wave formula. Pass floor(t / interval) as seed and the wave rotates automatically — no state, no timers, no extra variables.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

const INTERVAL = 1.5; // switch wave every 1.5 s

function setup() {
  createCanvas(600, 450);
  noFill();
}

function draw() {
  background(245);
  const t      = frameCount * 0.01;
  const period = floor(t / INTERVAL);

  noFill(); stroke(0); strokeWeight(1.5);
  beginShape();
  for (let x = 0; x <= width; x += 3) {
    vertex(x, height / 2 + Waves.wave(x * 0.02, {
      seed:      period,   // new wave each interval
      t:         t,
      amplitude: 80
    }));
  }
  endShape();

  noStroke(); fill(0, 0, 0, 80);
  textFont('monospace'); textSize(9); textAlign(LEFT);
  text('seed ' + period + '  →  ' +
    Waves.data[period % Waves.count].name, 10, 16);
}
API: Waves.list() · wave: N

All 34 waves

Waves.list() returns every formula with its name and index. Select by name (wave: 'noise') or directly by index (wave: 3).

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

function setup() {
  createCanvas(600, 560);
  noFill(); stroke(0); strokeWeight(1.2);
  textFont('monospace'); textSize(8);
}

function draw() {
  background(238);
  const t    = frameCount * 0.012;
  const COLS = 5, ROWS = 7;
  const cw    = width / COLS;
  const rh    = height / ROWS;
  const all   = Waves.list(); // 34 entries
  const limit = cw / 2 - 6;

  for (let i = 0; i < COLS * ROWS; i++) {
    if (i >= all.length) break;
    const col = i % COLS;
    const row = Math.floor(i / COLS);
    const x0 = col * cw, y0 = row * rh;

    noFill(); stroke(0,0,0,22);
    strokeWeight(0.5); rect(x0, y0, cw, rh);

    stroke(0); strokeWeight(1.2);
    beginShape();
    for (let y = y0+18; y < y0+rh-4; y += 2) {
      const raw = Waves.wave(y * 0.03, { wave: i, t, amplitude: limit });
      vertex(x0 + cw/2 + Math.tanh(raw / limit) * limit, y);
    }
    endShape();

    noStroke(); fill(0,0,0,100);
    textAlign(LEFT);
    text('#'+i+' '+all[i].name, x0+5, y0+13);
    stroke(0); noFill();
  }
}

waves: noise · grow random · fuzzy peak sine · fuzzy pulse

Noise & random waves

Several formulas produce structurally different randomness — smooth noise, growing random values, fuzzy peaks, irregular pulses. Same seed + same x always returns the same number: deterministic but visually unpredictable.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

function setup() {
  createCanvas(600, 400);
  noFill();
}

function draw() {
  background(245);
  const t = frameCount * 0.008;
  const types = [
    'noise',
    'grow random',
    'fuzzy peak sine',
    'fuzzy pulse'
  ];

  for (let i = 0; i < types.length; i++) {
    const wave = types[i];
    const y0 = map(i, 0, 3, 70, height - 70);
    stroke(0, 0, 0, 200 - i * 30);
    strokeWeight(1.4);
    beginShape();
    for (let x = 0; x <= width; x += 2) {
      vertex(x, y0 + Waves.wave(x * 0.018, { wave, t, amplitude: 38, seed: 3 }));
    }
    endShape();

    noStroke(); fill(0, 0, 0, 80);
    textFont('monospace'); textSize(9);
    textAlign(LEFT);
    text(wave, 8, y0 - 18);
    stroke(0); noFill();
  }
}
option: mode: 'wild' · unpredictability: 0–1

Wild mode

Any wave can run in wild mode. Left: smooth. Right: mode: 'wild' with unpredictability: 0.65. Wild mode injects position-varying noise into frequency, phase and amplitude simultaneously.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

function setup() {
  createCanvas(600, 400);
  noStroke();
}

function draw() {
  background(238);
  const t = frameCount * 0.014;

  for (let y = 4; y < height; y += 6) {
    // Left — smooth
    const xs = Waves.wave(y * 0.014, {
      wave:  'mountain peaks',
      t,
      range: [8, width / 2 - 8]
    });

    // Right — same wave, wild mode
    const xw = Waves.wave(y * 0.014, {
      wave:             'mountain peaks',
      t,
      mode:             'wild',
      unpredictability:  0.65,
      range:            [width / 2 + 8, width - 8]
    });

    fill(0, 0, 0, 170);
    circle(xs, y, 4);
    circle(xw, y, 4);
  }

  stroke(0,0,0,35); strokeWeight(1);
  line(width/2, 0, width/2, height);
  noStroke(); fill(0,0,0,90);
  textFont('monospace'); textSize(9);
  textAlign(CENTER);
  text('smooth', width/4, 14);
  text('wild  (0.65)', 3*width/4, 14);
}

API: Waves.createSampler(options)

CreateSampler

Configure once, sample hundreds of times per frame. Two samplers with different wave shapes drive X and Y — each phase step traces an orbit unique to that combination.

copy-paste sketch (instance mode)
// Needs p5.waves.js loaded after p5.js

new p5(function(p) {
  let sx, sy;
  const N = 90;

  p.setup = function() {
    p.createCanvas(500, 500);
    p.noStroke();

    sx = Waves.createSampler({
      wave:  'classic sine',
      seed:  3,
      range: [-200, 200]
    });
    sy = Waves.createSampler({
      wave:  'wobble sine',
      seed:  9,
      range: [-200, 200]
    });
  };

  p.draw = function() {
    p.background(245, 245, 245, 18);
    const t = p.frameCount * 0.012;

    for (let i = 0; i < N; i++) {
      const phase = (i / N) * p.TWO_PI * 2;
      const x = p.width  / 2 + sx.sample(phase, t);
      const y = p.height / 2 + sy.sample(phase + 0.8, t);
      p.fill(0, 0, 0, p.map(i, 0, N, 30, 200));
      p.circle(x, y, 4);
    }
  };
});

API: Waves.createGrid(cols, rows, options)

CreateGrid

One call generates a full 2D grid per frame. With threshold it returns a Uint8Array of 0/1 values — a binary pattern you can render any way you like.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

const COLS = 42, ROWS = 42;
let grid;

function setup() {
  createCanvas(500, 500);
  noStroke();

  grid = Waves.createGrid(COLS, ROWS, {
    waveRow:   'smooth solid sine',
    waveCol:   'ramp up sine',
    threshold:  0,    // → Uint8Array (0/1)
    speed:      1
  });
}

function draw() {
  background(238);
  const cells = grid.sample(frameCount * 0.004);
  const cw    = width  / COLS;
  const ch    = height / ROWS;

  for (let row = 0; row < ROWS; row++) {
    for (let col = 0; col < COLS; col++) {
      const v = cells[row * COLS + col];
      fill(v === 1 ? 15 : 225);
      rect(col * cw, row * ch, cw - 1, ch - 1);
    }
  }
}
API: Waves.createSampler() × 4 — particles

Flow fields

Four samplers — each a different wave — steer 150 particles. Configure once, sample every particle on every frame. Swap the wave names to change the flow pattern entirely.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

let samplers, particles = [];

function setup() {
  createCanvas(600, 400);
  noStroke();

  samplers = [
    Waves.createSampler({ wave: 'classic sine', seed:  0, range: [-2, 2] }),
    Waves.createSampler({ wave: 'wobble sine',  seed: 11, range: [-2, 2] }),
    Waves.createSampler({ wave: 'pulse',        seed: 22, range: [-2, 2] }),
    Waves.createSampler({ wave: 'meta sine',    seed: 33, range: [-2, 2] }),
  ];

  for (let i = 0; i < 150; i++) {
    particles.push({
      x: random(width),
      y: random(height),
      s: i % 4
    });
  }
}

function draw() {
  background(245, 245, 245, 22);
  const t = frameCount * 0.014;

  for (const p of particles) {
    p.x += samplers[p.s].sample(p.y * 0.009, t) * 2.5;
    p.y += samplers[(p.s+2)%4].sample(p.x * 0.009, t) * 2.5;
    if (p.x < 0) p.x += width;
    if (p.x > width)  p.x -= width;
    if (p.y < 0) p.y += height;
    if (p.y > height) p.y -= height;
    fill(0, 0, 0, 160);
    circle(p.x, p.y, 3);
  }
}

feature: t is just a number

Time is a number

There is no built-in clock. Pass any number as t — move the mouse to scrub through time. Freeze it, rewind it, slow it down for frame captures.

copy-paste sketch · move mouse left–right
// Needs p5.waves.js loaded after p5.js

const LAYERS      = 16;
const WAVE_WINDOW = 6;

function setup() {
  createCanvas(600, 400);
  noFill();
}

function draw() {
  background(238);

  // t from mouseX — no auto-advance
  const tBase = map(mouseX, 0, width, 0, 8);

  for (let i = 0; i < LAYERS; i++) {
    // Each layer gets its own slice of time
    const t  = tBase + (i / LAYERS) * WAVE_WINDOW;
    const y0 = map(i, 0, LAYERS-1, 30, height-30);

    stroke(0, 0, 0, map(i, 0, LAYERS-1, 210, 30));
    strokeWeight(map(i, 0, LAYERS-1, 2, 0.5));

    beginShape();
    for (let x = 0; x <= width; x += 3) {
      const y = y0 + Waves.wave(x * 0.015, {
        wave: 'classic sine', t, amplitude: 55
      });
      vertex(x, y);
    }
    endShape();
  }

  noStroke(); fill(0,0,0,80);
  textFont('monospace'); textSize(10);
  textAlign(LEFT);
  text('t = ' + map(mouseX,0,width,0,8).toFixed(2), 10, 16);
}

output: hue · saturation — colorMode(HSB)

Wave as colour

Wave output mapped to hue and saturation in HSB mode. No x/y position involved — the wave value is the colour. Two independent waves create a drifting interference in the spectrum.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

function setup() {
  createCanvas(600, 400);
  colorMode(HSB, 360, 100, 100);
  noStroke();
}

function draw() {
  const t = frameCount * 0.008;
  for (let y = 0; y < height; y += 2) {
    const hue = Waves.wave(y * 0.015, {
      wave:  'classic sine',
      t,
      range: [0, 360]   // → hue
    });
    const sat = Waves.wave(y * 0.015, {
      wave:  'wobble sine',
      t:     t * 0.6,
      range: [30, 100]  // → saturation
    });
    fill(hue, sat, 88);
    rect(0, y, width, 2);
  }
}

output: oscillator frequency — Web Audio API

Wave as sound

Three createSampler() calls with Hz ranges drive Web Audio oscillators. A moving cursor reads the wave as a score — what you see is what you hear. No p5.sound needed.

key parts — full runnable sketch → example 13
// Needs p5.waves.js loaded after p5.js

const samplers = [
  Waves.createSampler({ wave: 'classic sine', range: [110, 330] }),
  Waves.createSampler({ wave: 'triangle',     range: [220, 550] }),
  Waves.createSampler({ wave: 'stepped sine', range: [330, 770] }),
];
let audioCtx, oscs = [], running = false;

function mousePressed() {
  if (!running) {
    audioCtx = new AudioContext();
    oscs = [];
    for (let i = 0; i < samplers.length; i++) {
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      gain.gain.value = 0.05;
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      oscs.push(osc);
    }
    running = true;
  } else {
    for (const o of oscs) o.stop();
    audioCtx.close(); oscs = []; running = false;
  }
}

function draw() {
  const cursor = (frameCount * 2) % width;
  const t = frameCount * 0.012;
  // ... draw wave paths, update osc.frequency.value
  const freq = samplers[i].sample(cursor * 0.018, t);
  if (running) oscs[i].frequency.value = freq;
}

output: textSize · vertical lift · opacity

Wave as type

One sampler per letter drives font size independently. A second wave controls vertical lift; a third drives opacity. Same idea as position output — but applied to typographic properties.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

const WORD = 'WAVES';
const sizeSamplers = [];
for (let i = 0; i < 5; i++) {
  sizeSamplers.push(Waves.createSampler({ seed: i * 7, range: [28, 96] }));
}

function setup() {
  createCanvas(600, 400);
  textAlign(CENTER, CENTER);
  textFont('monospace');
  noStroke();
}

function draw() {
  background(245);
  const t = frameCount * 0.018;
  const sizes = [];
  for (let i = 0; i < WORD.length; i++) sizes.push(sizeSamplers[i].sample(i * 0.6, t));
  let totalW = 0;
  for (let i = 0; i < sizes.length; i++) totalW += sizes[i] * 0.72;
  let x = (width - totalW) / 2 + sizes[0] * 0.36;

  for (let i = 0; i < WORD.length; i++) {
    const sz    = sizes[i];
    const lift  = Waves.wave(i * 0.9 + t * 0.6,  { seed: i+20, range: [-38, 38] });
    const alpha = Waves.wave(i * 1.1 + t * 0.35, { seed: i+40, range: [80, 255] });
    textSize(sz);
    fill(0, 0, 0, alpha);
    text(WORD[i], x, height / 2 + lift);
    x += sz * 0.72;
  }
}

output: alpha — no movement, no size change

Wave as opacity

Two overlapping grids — same positions, same sizes — only alpha moves. Two waves at different speeds create interference patterns that appear and dissolve without anything changing shape or place.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

const COLS = 11, ROWS = 11;

function setup() {
  createCanvas(600, 600);
  noStroke();
}

function draw() {
  background(245);
  const t  = frameCount * 0.01;
  const cw = width  / COLS;
  const ch = height / ROWS;

  for (let r = 0; r < ROWS; r++) {
    for (let c = 0; c < COLS; c++) {
      const cx = (c + 0.5) * cw;
      const cy = (r + 0.5) * ch;

      const a1 = Waves.wave(c*0.35 + r*0.28, {
        wave: 'classic sine', t, range: [0, 220]
      });
      const a2 = Waves.wave(c*0.28 - r*0.35, {
        wave: 'triangle', t: t*1.6, range: [0, 180]
      });

      fill(0, a1);
      circle(cx, cy, cw * 0.72);
      fill(30, 80, 180, a2);
      circle(cx, cy, cw * 0.38);
    }
  }
}

reference: all 34 formulas at once — wave: i

Wave chart

All 34 formulas rendered as animated bars simultaneously. Bar height = current output of each wave at the same time input. Index numbers match { wave: i } — use as a live reference when choosing a formula.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

function setup() {
  createCanvas(600, 400);
  noStroke();
  textAlign(CENTER, TOP);
}

function draw() {
  background(245);
  const n    = Waves.count;  // 34
  const t    = frameCount * 0.016;
  const barW = width / n;
  const midY = height * 0.54;
  const amp  = height * 0.40;

  colorMode(HSB, n, 100, 100);
  for (let i = 0; i < n; i++) {
    const v = Waves.wave(t * 28, {
      wave:  i,           // direct index
      range: [-amp, amp]
    });
    const h = abs(v);
    const y = v < 0 ? midY : midY - h;

    fill(i, 65, 78);
    rect(i * barW + 1, y, barW - 2, h);
  }
  colorMode(RGB, 255);
}

WEBGL · 3D · wave cycling

3D Wave Volume

16×16×16 point lattice (4 096 points). Two Waves.wave() calls — one along X, one along Z — define a floating wave surface. Points near the surface glow; the formula cycles through all 34 types every 2 seconds.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

var N = 16, SPACING = 22, CYCLE = 60;

function setup() {
  createCanvas(460, 460, WEBGL);
  frameRate(30);
  noFill();
  stroke(120, 160, 255);
  strokeWeight(4);
}

function draw() {
  background(12);
  rotateX(-0.5);
  rotateY(frameCount * 0.006);
  var t     = frameCount * 0.008;
  var half  = (N - 1) * SPACING / 2;
  var waveX = Math.floor(frameCount / CYCLE) % Waves.count;
  var waveZ = (waveX + 1) % Waves.count;
  beginShape(POINTS);
  for (var xi = 0; xi < N; xi++) {
    for (var zi = 0; zi < N; zi++) {
      var wv = Waves.wave(xi * 0.38, { wave: waveX, t: t,        amplitude: 7.5 })
             + Waves.wave(zi * 0.38, { wave: waveZ, t: t * 0.85, amplitude: 7.5 });
      var yi = Math.max(0, Math.min(N - 1, Math.round(N / 2 + wv / 2)));
      vertex(-half + xi * SPACING, -half + yi * SPACING, -half + zi * SPACING);
    }
  }
  endShape();
}
