API: Waves.wave(y, options)

The core function

One call returns one number. Pass a position and options — wave shape, time, amplitude, frequency. Use the result as offset, size, colour value, or anything else.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

function setup() {
  createCanvas(600, 400);
  noFill();
}

function draw() {
  background(245);
  const t = frameCount * 0.01;
  const waves = [
    'classic sine', 'triangle',
    'mountain peaks', 'noise'
  ];

  waves.forEach((wave, i) => {
    const y0 = map(i, 0, 3, 70, height - 70);
    stroke(0, 0, 0, map(i, 0, 3, 220, 80));
    strokeWeight(1.5);
    beginShape();
    for (let x = 0; x <= width; x += 3) {
      const y = y0 + Waves.wave(x * 0.015, {
        wave, t, amplitude: 45
      });
      vertex(x, y);
    }
    endShape();
  });
}

API: Waves.list() · wave: N

All 34 formulas

Waves.list() returns every formula with its name and index. Select by name (wave: 'noise') or directly by index (wave: 3).

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

function setup() {
  createCanvas(600, 560);
  noFill(); stroke(0); strokeWeight(1.2);
  textFont('monospace'); textSize(8);
}

function draw() {
  background(238);
  const t    = frameCount * 0.012;
  const COLS = 5, ROWS = 7;
  const cw   = width / COLS;
  const rh   = height / ROWS;
  const all  = Waves.list(); // 34 entries

  for (let i = 0; i < COLS * ROWS; i++) {
    if (i >= all.length) break;
    const col = i % COLS;
    const row = Math.floor(i / COLS);
    const x0 = col * cw, y0 = row * rh;

    noFill(); stroke(0,0,0,22);
    strokeWeight(0.5); rect(x0, y0, cw, rh);

    stroke(0); strokeWeight(1.2);
    beginShape();
    for (let y = y0+18; y < y0+rh-4; y += 2) {
      const x = Waves.wave(y * 0.03, {
        wave: i, t,    // i = direct index
        range: [x0+6, x0+cw-6]
      });
      vertex(x, y);
    }
    endShape();

    noStroke(); fill(0,0,0,100);
    textAlign(LEFT);
    text('#'+i+' '+all[i].name, x0+5, y0+13);
    stroke(0); noFill();
  }
}

Noise & random waves

Several formulas produce structurally different randomness — smooth noise, growing random values, fuzzy peaks, irregular pulses. Same seed + same x always returns the same number: deterministic but visually unpredictable.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

function setup() {
  createCanvas(600, 400);
  noFill();
}

function draw() {
  background(245);
  const t = frameCount * 0.008;
  const types = [
    'noise',
    'grow random',
    'fuzzy peak sine',
    'fuzzy pulse'
  ];

  types.forEach((wave, i) => {
    const y0 = map(i, 0, 3, 70, height - 70);
    stroke(0, 0, 0, 200 - i * 30);
    strokeWeight(1.4);
    beginShape();
    for (let x = 0; x <= width; x += 2) {
      const y = y0 + Waves.wave(x * 0.018, {
        wave,
        t,
        amplitude: 38,
        seed: 3
      });
      vertex(x, y);
    }
    endShape();

    noStroke(); fill(0, 0, 0, 80);
    textFont('monospace'); textSize(9);
    textAlign(LEFT);
    text(wave, 8, y0 - 18);
    stroke(0); noFill();
  });
}
option: mode: 'wild' · unpredictability: 0–1

Wild mode

Any wave can run in wild mode. Left: smooth. Right: mode: 'wild' with unpredictability: 0.65. Wild mode injects position-varying noise into frequency, phase and amplitude simultaneously.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

function setup() {
  createCanvas(600, 400);
  noStroke();
}

function draw() {
  background(238);
  const t = frameCount * 0.014;

  for (let y = 4; y < height; y += 6) {
    // Left — smooth
    const xs = Waves.wave(y * 0.014, {
      wave:  'mountain peaks',
      t,
      range: [8, width / 2 - 8]
    });

    // Right — same wave, wild mode
    const xw = Waves.wave(y * 0.014, {
      wave:             'mountain peaks',
      t,
      mode:             'wild',
      unpredictability:  0.65,
      range:            [width / 2 + 8, width - 8]
    });

    fill(0, 0, 0, 170);
    circle(xs, y, 4);
    circle(xw, y, 4);
  }

  stroke(0,0,0,35); strokeWeight(1);
  line(width/2, 0, width/2, height);
  noStroke(); fill(0,0,0,90);
  textFont('monospace'); textSize(9);
  textAlign(CENTER);
  text('smooth', width/4, 14);
  text('wild  (0.65)', 3*width/4, 14);
}

API: Waves.createSampler(options)

CreateSampler

Configure once, sample hundreds of times per frame. Two samplers with different wave shapes drive X and Y — each phase step traces an orbit unique to that combination.

copy-paste sketch (instance mode)
// Needs p5.waves.js loaded after p5.js

new p5(p => {
  let sx, sy;
  const N = 90;

  p.setup = () => {
    p.createCanvas(500, 500);
    p.noStroke();

    sx = Waves.createSampler({
      wave:  'classic sine',
      seed:  3,
      range: [-200, 200]
    });
    sy = Waves.createSampler({
      wave:  'wobble',
      seed:  9,
      range: [-200, 200]
    });
  };

  p.draw = () => {
    p.background(245, 245, 245, 18);
    const t = p.frameCount * 0.012;

    for (let i = 0; i < N; i++) {
      const phase = (i / N) * p.TWO_PI * 2;
      const x = p.width  / 2 + sx.sample(phase, t);
      const y = p.height / 2 + sy.sample(phase + 0.8, t);
      p.fill(0, 0, 0, p.map(i, 0, N, 30, 200));
      p.circle(x, y, 4);
    }
  };
});

API: Waves.createGrid(cols, rows, options)

CreateGrid

One call generates a full 2D grid per frame. With threshold it returns a Uint8Array of 0/1 values — a binary pattern you can render any way you like.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

const COLS = 42, ROWS = 42;
let grid;

function setup() {
  createCanvas(500, 500);
  noStroke();

  grid = Waves.createGrid(COLS, ROWS, {
    waveRow:   'smooth solid sine',
    waveCol:   'ramp up sine',
    threshold:  0,    // → Uint8Array (0/1)
    speed:      1
  });
}

function draw() {
  background(238);
  const cells = grid.sample(frameCount * 0.004);
  const cw    = width  / COLS;
  const ch    = height / ROWS;

  for (let row = 0; row < ROWS; row++) {
    for (let col = 0; col < COLS; col++) {
      const v = cells[row * COLS + col];
      fill(v === 1 ? 15 : 225);
      rect(col * cw, row * ch, cw - 1, ch - 1);
    }
  }
}
API: Waves.createSampler() × 4 — particles

Flow fields

Four samplers — each a different wave — steer 150 particles. Configure once, sample every particle on every frame. Swap the wave names to change the flow pattern entirely.

copy-paste sketch
// Needs p5.waves.js loaded after p5.js

let samplers = [], particles = [];

function setup() {
  createCanvas(600, 400);
  noStroke();

  ['classic sine', 'wobble',
   'pulse', 'plasma'].forEach((w, i) => {
    samplers.push(Waves.createSampler({
      wave: w, seed: i * 11, range: [-2, 2]
    }));
  });

  for (let i = 0; i < 150; i++) {
    particles.push({
      x: random(width),
      y: random(height),
      s: i % 4
    });
  }
}

function draw() {
  background(245, 245, 245, 22);
  const t = frameCount * 0.014;

  for (const p of particles) {
    p.x += samplers[p.s].sample(p.y * 0.009, t) * 2.5;
    p.y += samplers[(p.s+2)%4].sample(p.x * 0.009, t) * 2.5;
    if (p.x < 0) p.x += width;
    if (p.x > width)  p.x -= width;
    if (p.y < 0) p.y += height;
    if (p.y > height) p.y -= height;
    fill(0, 0, 0, 160);
    circle(p.x, p.y, 3);
  }
}

feature: t is just a number

Time is a number

There is no built-in clock. Pass any number as t — move the mouse to scrub through time. Freeze it, rewind it, slow it down for frame captures.

copy-paste sketch · move mouse left–right
// Needs p5.waves.js loaded after p5.js

const LAYERS = 18;

function setup() {
  createCanvas(600, 400);
  noFill();
}

function draw() {
  background(238);

  // t from mouseX — no auto-advance
  const tBase = map(mouseX, 0, width, 0, 8);

  for (let i = 0; i < LAYERS; i++) {
    // Each layer gets its own slice of time
    const t  = tBase + (i / LAYERS) * 3;
    const y0 = map(i, 0, LAYERS-1, 30, height-30);

    stroke(0, 0, 0, map(i, 0, LAYERS-1, 210, 30));
    strokeWeight(map(i, 0, LAYERS-1, 2, 0.5));

    beginShape();
    for (let x = 0; x <= width; x += 3) {
      const y = y0 + Waves.wave(x * 0.015, {
        wave: 'classic sine', t, amplitude: 28
      });
      vertex(x, y);
    }
    endShape();
  }

  noStroke(); fill(0,0,0,80);
  textFont('monospace'); textSize(10);
  textAlign(LEFT);
  text('t = ' + map(mouseX,0,width,0,8).toFixed(2), 10, 16);
}
