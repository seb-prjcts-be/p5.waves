system / assembly

From One Wave To A Poster

The curation engine works when the page stops treating motion as decoration. Pools, samplers, modules, and readouts become one authored system.

Live assembly preview · ribbons + grid + title + atmosphere
start: pool -> role -> module
end product: one engine, many surfaces
start Stop thinking in single effects

The key shift is asking which family of wave behavior belongs to each layer, instead of asking how to animate one object.

build Assign roles, not values

Pools curate tone, samplers curate responsibility, and modules reveal the choices in public through layout, type, and proof panels.

end product One engine, many surfaces

The poster feels designed because motion, structure, and interface all come from the same logic instead of unrelated effects.

const ARRAY_GROUP = [
  'classic sine',
  'mountain peaks',
  'triangle sine',
  'smooth solid sine',
  'meta sine'
];

const GROUPS = [
  { key: 'gentle', group: 'gentle', color: '#174cff' },
  { key: 'harsh', group: 'harsh', color: '#ff3b2f' },
  { key: 'all', group: 'all', color: '#d7ff22' },
  { key: 'array', group: ARRAY_GROUP, color: '#ff4fb3' }
];

let typeSampler;
let groupSamplers = [];
let gridSamplers = [];

function setup() {
  createCanvas(960, 540);
  noStroke();
  textFont('Oswald');

  typeSampler = Waves.createSampler({
    group: ARRAY_GROUP,
    shift: true,
    shiftInterval: 6,
    shiftDuration: 1.5,
    range: [-1, 1],
    frequency: 0.72
  });

  groupSamplers = GROUPS.map(function(preset, i) {
    return Waves.createSampler({
      group: preset.group,
      shift: true,
      shiftInterval: 3.4 + i * 0.35,
      shiftDuration: 1.1,
      range: [-1, 1],
      frequency: 0.7 + i * 0.08
    });
  });

  gridSamplers = GROUPS.map(function(preset, i) {
    return Waves.createGrid(24, 12, {
      group: preset.group,
      threshold: preset.key === 'harsh' ? 0.12 : 0,
      speed: 0.42 + i * 0.08
    });
  });
}

function draw() {
  background('#f4efe5');
  const t = millis() / 1000;
  const active = floor(t / 4.8) % GROUPS.length;

  for (let i = 0; i < GROUPS.length; i++) {
    const y = map(i, 0, GROUPS.length - 1, height * 0.24, height * 0.78);
    const amp = 24 + abs(groupSamplers[i].sample(i * 0.2, t)) * 42;

    fill(color(GROUPS[i].color + (i === active ? 'aa' : '66')));
    beginShape();
    for (let x = -20; x <= width + 20; x += 16) {
      vertex(x, y + Waves.wave(x * 0.014, {
        group: GROUPS[i].group,
        shift: true,
        shiftInterval: 3.6,
        shiftDuration: 1.05,
        t: t + i * 0.35,
        amplitude: amp,
        frequency: 0.76
      }));
    }
    for (let x = width + 20; x >= -20; x -= 16) {
      vertex(x, y + 28);
    }
    endShape(CLOSE);
  }

  const cells = gridSamplers[active].sample(t);
  const gx = width - 270;
  const gy = 48;
  const cw = 240 / 24;
  const ch = 120 / 12;

  fill('#080808');
  rect(gx - 6, gy - 6, 252, 132);
  fill('#f4efe5');
  rect(gx, gy, 240, 120);

  for (let row = 0; row < 12; row++) {
    for (let col = 0; col < 24; col++) {
      const on = cells[row * 24 + col] === 1;
      fill(on ? GROUPS[active].color : '#ece5d7');
      rect(gx + col * cw, gy + row * ch, cw - 1, ch - 1);
    }
  }

  textSize(84);
  let x = 42;
  for (let i = 0; i < 'p5.waves'.length; i++) {
    const ch = 'p5.waves'[i];
    const dy = typeSampler.sample(i * 0.23, t) * 8;
    fill(i === 3 || i === 5 ? GROUPS[active].color : '#080808');
    text(ch, x, 120 + dy);
    x += textWidth(ch) * 0.97;
  }

  fill('#080808');
  textSize(30);
  text('AS A CURATION ENGINE', 42, 205);

  fill('#080808');
  rect(42, height - 60, 280, 24);
  fill(GROUPS[active].color);
  rect(42, height - 60, 160, 24);
}

01 / pool curation

Curate The Pool

This is the editorial backbone. Instead of letting every layer access every wave, you decide which behavior family belongs to which job.

start Name the voices

gentle is readable, harsh ruptures, all stays broad, and the array pool is hand-picked for taste.

build Bind color to character

Each group gets a visible identity: a label, a pool, and a color. The selection logic becomes visual direction.

end product Rotation without chaos

The active pool can change over time, but the sketch still speaks in a controlled voice because each layer stays in its family.

preview: four curated families as live rows
result: variation with authorship
const ARRAY_GROUP = [

  'mountain peaks',
  'triangle sine',
  'smooth solid sine',
  'meta sine'
];

const GROUP_PRESETS = [
  { key: 'gentle', group: 'gentle', color: '#174cff' },
  { key: 'harsh', group: 'harsh', color: '#ff3b2f' },
  { key: 'all', group: 'all', color: '#d7ff22' },
  { key: 'array', group: ARRAY_GROUP, color: '#ff4fb3' }
];

let groupSamplers = [];

function setup() {
  createCanvas(720, 420);
  noStroke();
  textFont('Oswald');

  groupSamplers = GROUP_PRESETS.map(function(preset, i) {
    return Waves.createSampler({
      group: preset.group,
      shift: true,
      shiftInterval: 3.2 + i * 0.4,
      shiftDuration: 1,
      range: [-1, 1],
      frequency: 0.65 + i * 0.08,
      mode: preset.key === 'harsh' ? 'wild' : 'stable',
      unpredictability: preset.key === 'harsh' ? 0.32 : 0
    });
  });
}

function draw() {
  background('#f4efe5');
  const t = millis() / 1000;
  const active = floor(t / 4.8) % GROUP_PRESETS.length;
  const left = 140;
  const rowH = 72;
  const gap = 16;

  for (let i = 0; i < GROUP_PRESETS.length; i++) {
    const preset = GROUP_PRESETS[i];
    const y = 38 + i * (rowH + gap);
    const amp = 12 + abs(groupSamplers[i].sample(i * 0.22, t)) * 28;

    stroke('#080808');
    strokeWeight(i === active ? 3 : 1);
    fill(i === active ? '#fffdfa' : '#f7f1e7');
    rect(24, y, width - 48, rowH);
    noStroke();

    fill('#080808');
    textSize(18);
    text(preset.key.toUpperCase(), 40, y + 22);

    fill(color(preset.color + 'aa'));
    beginShape();
    for (let x = left; x <= width - 32; x += 14) {
      const wave = Waves.wave(x * 0.016 + i * 0.3, {
        group: preset.group,
        shift: true,
        shiftInterval: 3.4,
        shiftDuration: 1,
        t: t + i * 0.3,
        amplitude: amp,
        frequency: 0.82
      });
      vertex(x, y + rowH * 0.48 + wave);
    }
    for (let x = width - 32; x >= left; x -= 14) {
      vertex(x, y + rowH - 12);
    }
    endShape(CLOSE);
  }
}

02 / sampler roles

Split The Responsibilities

Separate samplers are what make the composition feel layered. Motion, color, pulse, and type stop collapsing into one uniform behavior.

start Define layer jobs

List what the sketch needs: drift, pulse, color selection, and typographic wobble. Those are different roles.

build Tune each role separately

Readable motion uses gentle, pulse uses harsh, and typography uses the curated array. Each layer gets its own temperament.

end product Layered behavior

The page gains nuance because not everything responds the same way. The system reads as composed instead of globally animated.

preview: motion, color, pulse, type
result: four roles, four behaviors
let motionSampler;
let colorSampler;
let pulseSampler;
let typeSampler;

const ARRAY_GROUP = [
  'classic sine',
  'mountain peaks',
  'triangle sine',
  'smooth solid sine',
  'meta sine'
];

function setup() {
  createCanvas(720, 420);
  noStroke();
  textFont('Oswald');

  motionSampler = Waves.createSampler({
    group: 'gentle',
    shift: true,
    shiftInterval: 5.5,
    shiftDuration: 1.4,
    range: [-1, 1],
    frequency: 0.82
  });

  colorSampler = Waves.createSampler({
    group: 'all',
    shift: true,
    shiftInterval: 6.2,
    shiftDuration: 1.6,
    range: [0, 1],
    frequency: 0.56
  });

  pulseSampler = Waves.createSampler({
    group: 'harsh',
    shift: true,
    shiftInterval: 4,
    shiftDuration: 1.1,
    range: [0, 1],
    frequency: 1.05,
    mode: 'wild',
    unpredictability: 0.28
  });

  typeSampler = Waves.createSampler({
    group: ARRAY_GROUP,
    shift: true,
    shiftInterval: 6.8,
    shiftDuration: 1.5,
    range: [-1, 1],
    frequency: 0.7
  });
}

function draw() {
  background('#f4efe5');
  const t = millis() / 1000;
  const w = width * 0.42;
  const h = height * 0.34;
  const cells = [
    [32, 32, 'motion'],
    [width - w - 32, 32, 'color'],
    [32, height - h - 32, 'pulse'],
    [width - w - 32, height - h - 32, 'type']
  ];

  for (const cell of cells) {
    const x = cell[0];
    const y = cell[1];
    const label = cell[2];
    fill('#fffdfa');
    stroke('#080808');
    rect(x, y, w, h);
    noStroke();
    fill('#080808');
    textSize(18);
    text(label.toUpperCase(), x + 16, y + 24);

    if (label === 'motion') {
      const mx = x + w * 0.5 + motionSampler.sample(0.2, t) * w * 0.22;
      const my = y + h * 0.62 + motionSampler.sample(0.45, t + 1) * h * 0.18;
      fill('#174cff');
      circle(mx, my, 42);
    }

    if (label === 'color') {
      const v = colorSampler.sample(0.3, t);
      const shades = ['#174cff', '#ff3b2f', '#d7ff22', '#ff4fb3', '#00c7ff'];
      const idx = constrain(floor(v * shades.length), 0, shades.length - 1);
      fill(shades[idx]);
      rect(x + 16, y + 40, (w - 32) * v, h - 56);
    }

    if (label === 'pulse') {
      for (let i = 0; i < 8; i++) {
        const power = pulseSampler.sample(i * 0.18, t);
        fill(i % 2 === 0 ? '#ff3b2f' : '#080808');
        rect(x + 16 + i * 22, y + h - 18 - power * 72, 14, power * 72);
      }
    }

    if (label === 'type') {
      let tx = x + 18;
      textSize(42);
      for (let i = 0; i < 4; i++) {
        const ch = 'WAVE'[i];
        const dy = typeSampler.sample(i * 0.25, t) * 8;
        fill(i === 1 ? '#ff4fb3' : '#080808');
        text(ch, tx, y + h * 0.62 + dy);
        tx += textWidth(ch) * 0.95;
      }
    }
  }
}

03 / wave ribbons

Turn Waves Into Material

A line describes. A ribbon divides space. This is where the library starts behaving like layout instead of motion garnish.

start Sample a line

Use Waves.wave() across the x-axis to get the upper contour of the future form.

build Close it as a band

Run the lower edge back with a vertical offset. The moment the shape closes, it becomes printable material.

end product Editorial strips

The final bands split the composition, carry tone, and push energy without abandoning the page structure.

preview: line -> ribbon -> band
result: wave as layout material
function drawRibbon(xOffset, y, thickness, amp, group, tint) {
  fill(color(tint));
  beginShape();
  for (let x = -20; x <= width + 20; x += 14) {
    vertex(x, y + Waves.wave(x * 0.015 + xOffset, {
      group: group,
      shift: true,
      shiftInterval: 3.6,
      shiftDuration: 1.05,
      t: millis() / 1000,
      amplitude: amp,
      frequency: 0.78
    }));
  }
  for (let x = width + 20; x >= -20; x -= 14) {
    vertex(x, y + thickness);
  }
  endShape(CLOSE);
}

function setup() {
  createCanvas(760, 420);
  noStroke();
}

function draw() {
  background('#f4efe5');

  stroke('#080808');
  noFill();
  beginShape();
  for (let x = -20; x <= width + 20; x += 14) {
    vertex(x, 90 + Waves.wave(x * 0.015, {
      group: 'gentle',
      shift: true,
      shiftInterval: 3.5,
      shiftDuration: 1,
      t: millis() / 1000,
      amplitude: 16,
      frequency: 0.82
    }));
  }
  endShape();

  noStroke();
  drawRibbon(0.2, 170, 26, 22, 'gentle', '#174cff88');
  drawRibbon(0.6, 275, 54, 46, 'harsh', '#ff3b2faa');

  fill('#080808');
  textFont('Oswald');
  textSize(18);
  text('LINE', 24, 62);
  text('FILLED RIBBON', 24, 140);
  text('EDITORIAL BAND', 24, 242);
}

const GROUP_PRESETS = [
  { key: 'gentle', group: 'gentle', color: '#174cff' },
  { key: 'harsh', group: 'harsh', color: '#ff3b2f' },
  { key: 'all', group: 'all', color: '#d7ff22' }
];

let grids = [];

function setup() {
  createCanvas(720, 420);
  noStroke();
  textFont('Oswald');

  grids = GROUP_PRESETS.map(function(preset, i) {
    return Waves.createGrid(24, 12, {
      group: preset.group,
      threshold: preset.key === 'harsh' ? 0.12 : 0,
      speed: 0.42 + i * 0.08,
      seed: 700 + i * 20
    });
  });
}

function draw() {
  background('#f4efe5');
  const t = millis() / 1000;
  const active = floor(t / 5) % GROUP_PRESETS.length;
  const cells = grids[active].sample(t);
  const x = 90;
  const y = 56;
  const cols = 24;
  const rows = 12;
  const cw = 540 / cols;
  const ch = 270 / rows;

  fill('#080808');
  rect(x - 8, y - 8, 556, 286);
  fill('#f4efe5');
  rect(x, y, 540, 270);

  for (let row = 0; row < rows; row++) {
    for (let col = 0; col < cols; col++) {
      const on = cells[row * cols + col] === 1;
      fill(on ? GROUP_PRESETS[active].color : '#ece5d7');
      rect(x + col * cw, y + row * ch, cw - 1, ch - 1);
    }
  }

  fill('#080808');
  textSize(18);
  text('CREATEGRID / GROUP: ' + GROUP_PRESETS[active].key.toUpperCase(), 90, 360);
}

05 / kinetic typography

Make The Title Breathe

The type works because the movement is sampled and constrained. The word stays typographic first, live second.

start Keep the word intact

Draw the title letter by letter while preserving the baseline and spacing logic of the word.

build Sample each letter

Offset every glyph with the same sampler at a different sample position so the letters move as a family.

end product Alive, not unstable

The final word drifts just enough to show life without breaking its own legibility.

preview: one word, offset per letter
result: typography that breathes
const ARRAY_GROUP = [
  'classic sine',
  'mountain peaks',
  'triangle sine',
  'smooth solid sine',
  'meta sine'
];

let typeSampler;
let colorSampler;

function setup() {
  createCanvas(760, 300);
  noStroke();
  textFont('Oswald');

  typeSampler = Waves.createSampler({
    group: ARRAY_GROUP,
    shift: true,
    shiftInterval: 6.6,
    shiftDuration: 1.5,
    range: [-1, 1],
    frequency: 0.72
  });

  colorSampler = Waves.createSampler({
    group: 'all',
    shift: true,
    shiftInterval: 6,
    shiftDuration: 1.4,
    range: [0, 1],
    frequency: 0.58
  });
}

function drawKineticWord(word, x, y, size, t) {
  textSize(size);
  let cursor = x;
  for (let i = 0; i < word.length; i++) {
    const ch = word[i];
    const wobble = typeSampler.sample(i * 0.24, t) * size * 0.05;
    const accent = colorSampler.sample(i * 0.18, t);
    const colors = ['#174cff', '#ff3b2f', '#d7ff22', '#ff4fb3'];
    const tone = colors[constrain(floor(accent * colors.length), 0, colors.length - 1)];
    fill(i === 3 || i === 5 ? tone : '#080808');
    text(ch, cursor, y + wobble);
    cursor += textWidth(ch) * 0.97;
  }
}

function draw() {
  background('#f4efe5');
  drawKineticWord('p5.waves', 34, 150, 110, millis() / 1000);
}

07 / ambient language

Let Vocabulary Become Atmosphere

The background field stays useful because it is still attached to the same engine as the hero layer. It adds context without turning generic.

start Choose a small vocabulary

Use words that belong to the library itself: wave names, behaviors, parameters, and descriptive tags.

build Drive drift and visibility

Motion moves the words, pulse decides whether they appear, and type drift keeps them from freezing in place.

end product Context without clutter

The words become ghost labels and marginalia behind the hero statement, not filler sprinkled on top.

preview: pulse-gated drifting vocabulary
result: ambient language field
const WORDS = ['pure sine', 'noise', 'modulo', 'quantized', 'peak valley'];

let motionSampler;
let pulseSampler;
let typeSampler;
let particles = [];

function setup() {
  createCanvas(760, 360);
  noStroke();
  textFont('Oswald');

  motionSampler = Waves.createSampler({
    group: 'gentle',
    shift: true,
    shiftInterval: 5.5,
    shiftDuration: 1.3,
    range: [-1, 1],
    frequency: 0.8
  });

  pulseSampler = Waves.createSampler({
    group: 'harsh',
    shift: true,
    shiftInterval: 4,
    shiftDuration: 1.1,
    range: [0, 1],
    frequency: 1.1,
    mode: 'wild',
    unpredictability: 0.25
  });

  typeSampler = Waves.createSampler({
    group: 'all',
    shift: true,
    shiftInterval: 6.2,
    shiftDuration: 1.4,
    range: [-1, 1],
    frequency: 0.68
  });

  for (let i = 0; i < 24; i++) {
    particles.push({
      seed: i,
      x: random(width),
      y: random(height),
      word: random(WORDS),
      size: random(13, 24)
    });
  }
}

function draw() {
  background('#f4efe5');
  const t = millis() / 1000;

  for (const item of particles) {
    const vx = motionSampler.sample(item.seed * 0.31, t) * 22;
    const vy = typeSampler.sample(item.seed * 0.19, t + 1.5) * 16;
    const a = pulseSampler.sample(item.seed * 0.13, t);
    if (a < 0.34) continue;

    fill(color('#080808' + hex(round(map(a, 0.34, 1, 40, 150)), 2)));
    textSize(item.size);
    text(item.word, item.x + vx, item.y + vy);
  }
}
